/// @dev Plain token with configurable decimals; tests mint freely.
contract Token {
    uint8 public immutable decimals;
    mapping(address => uint256) public balanceOf;

    constructor(uint8 decimals_) {
        decimals = decimals_;
    }

    function mint(address to, uint256 amount) external {
        balanceOf[to] += amount;
    }

    function transfer(address to, uint256 amount) external virtual returns (bool) {
        require(balanceOf[msg.sender] >= amount, "balance");
        balanceOf[msg.sender] -= amount;
        balanceOf[to] += amount;
        return true;
    }
}

/// @dev Returns false instead of reverting.
contract FalseToken is Token(6) {
    function transfer(address, uint256) external pure override returns (bool) {
        return false;
    }
}

/// @dev Calls back into the relay while it is being paid out.
contract HookToken is Token(6) {
    address public relay;
    bool public hookOn;

    function setRelay(address relay_) external {
        relay = relay_;
    }

    function setHook(bool on) external {
        hookOn = on;
    }

    function transfer(address to, uint256 amount) external override returns (bool) {
        if (hookOn) {
            hookOn = false;
            LatchRelay(payable(relay)).tick(1e6);
        }
        require(balanceOf[msg.sender] >= amount, "balance");
        balanceOf[msg.sender] -= amount;
        balanceOf[to] += amount;
        return true;
    }
}

/// @dev Stand-in for a TapeOut processor with the live X Layer TAP-02 semantics
///      (same as the processor reference checked against chain 196):
///      signal 0 = constant 0, signal 1 = constant 1, inputs at 2..2+nIn-1
///      (LSB-first from the input bytes), gate g output at 2+nIn+g, NAND only.
///      Output bit k = signal (last nOut + k), LSB-first.
contract NetlistProcessor {
    mapping(uint256 => bytes) public netlist;
    mapping(uint256 => uint32) public nInOf;
    mapping(uint256 => uint32) public nOutOf;

    function tape(uint256 id, bytes calldata net, uint32 nIn, uint32 nOut) external {
        netlist[id] = net;
        nInOf[id] = nIn;
        nOutOf[id] = nOut;
    }

    function eval(uint256 id, bytes calldata inputs) external view returns (bytes memory out) {
        bytes memory net = netlist[id];
        require(net.length > 0 && net.length % 7 == 0, "no circuit");
        uint256 nIn = nInOf[id];
        uint256 nOut = nOutOf[id];
        require(inputs.length == (nIn + 7) / 8, "bad inputs");
        uint256 n = net.length / 7;
        bool[] memory s = new bool[](2 + nIn + n);
        s[1] = true;
        for (uint256 i = 0; i < nIn; i++) {
            s[2 + i] = (uint8(inputs[i >> 3]) >> (i & 7)) & 1 == 1;
        }
        for (uint256 g = 0; g < n; g++) {
            uint256 o = g * 7;
            require(uint8(net[o]) == 0x00, "opcode");
            uint256 a = (uint256(uint8(net[o + 1])) << 16) | (uint256(uint8(net[o + 2])) << 8) | uint8(net[o + 3]);
            uint256 b = (uint256(uint8(net[o + 4])) << 16) | (uint256(uint8(net[o + 5])) << 8) | uint8(net[o + 6]);
            require(a < 2 + nIn + g && b < 2 + nIn + g, "wire");
            s[2 + nIn + g] = !(s[a] && s[b]);
        }
        out = new bytes((nOut + 7) / 8);
        uint256 base = 2 + nIn + n - nOut;
        for (uint256 k = 0; k < nOut; k++) {
            if (s[base + k]) out[k >> 3] |= bytes1(uint8(1 << (k & 7)));
        }
    }
}

/// @dev Processor that always answers the same thing (a liar or a wall).
contract ConstProcessor {
    bytes public answer;

    constructor(bytes memory answer_) {
        answer = answer_;
    }

    function eval(uint256, bytes calldata) external view returns (bytes memory) {
        return answer;
    }

    function netlist(uint256) external pure returns (bytes memory) {
        return "";
    }
}

/// @dev Minimal ERC-6551-style account: only the NFT holder can execute, and each
///      execute must pay the protocol fee in native coin, like TapeOut's containers.
contract Container {
    address public nftOwner;
    uint256 public constant PROTOCOL_FEE = 1.3e15;
    error ProtocolFeeTooLow(uint256 sent, uint256 needed);

    function setNftOwner(address o) external {
        nftOwner = o;
    }

    function execute(address to, uint256 value, bytes calldata data, uint8 operation)
        external
        payable
        returns (bytes memory)
    {
        require(msg.sender == nftOwner, "not NFT owner");
        if (msg.value < PROTOCOL_FEE) revert ProtocolFeeTooLow(msg.value, PROTOCOL_FEE);
        require(operation == 0, "call only");
        (bool ok, bytes memory ret) = to.call{value: value}(data); // fee stays in the container
        if (!ok) {
            assembly {
                revert(add(ret, 32), mload(ret))
            }
        }
        return ret;
    }
}

contract LatchRelayTest is Test {
    // The 13-gate netlist from src/lib/latch/circuit.ts (netlistHex()), TAP-02 numbering.
    bytes internal constant NETLIST =
        hex"00000002000002000000030000090000000a00000a0000000b0000040000000c00000c0000000d0000050000000e00000e0000000f0000060000001000001000000011000007000000120000120000001300000800000014000014";
    // The netlist before the fix: wires numbered as if inputs started at signal 0.
    bytes internal constant OLD_NETLIST =
        hex"000000000000000000000100000700000008000008000000090000020000000a00000a0000000b0000030000000c00000c0000000d0000040000000e00000e0000000f000005000000100000100000001100000600000012000012";
    uint256 internal constant CIRCUIT_ID = 7;
    uint256 internal constant U = 1e6; // 6-decimal USDT, as on X Layer
    uint256 internal constant EF = 1.3e15; // container execute fee, OKB

    address internal constant OWNER = address(0xC0FFEE);
    address internal constant KEEPER = address(0xCAFE);
    address internal constant SAVINGS = address(0x5A7E);
    address internal constant TREASURY = address(0x7BEA5);

    LatchRelay internal relay;
    Token internal token;
    NetlistProcessor internal processor;
    Container internal container;

    function setUp() public {
        vm.warp(1_800_000_000);
        token = new Token(6);
        processor = new NetlistProcessor();
        processor.tape(CIRCUIT_ID, NETLIST, 7, 1);
        container = new Container();
        relay = _deploy(address(processor), address(token));
        token.mint(address(container), 1000 * U);
        vm.prank(OWNER);
        relay.arm();
    }

    function _deploy(address proc, address tok) internal returns (LatchRelay r) {
        vm.prank(OWNER);
        r = new LatchRelay(proc, CIRCUIT_ID, address(container), tok, SAVINGS, TREASURY, KEEPER, 50 * U, 150 * U, 5 * U, EF);
        container.setNftOwner(address(r));
        vm.deal(address(r), 1 ether); // pre-funded for execute fees
    }

    // ------------------------------------------------------------ construction

    function testConstructorWiring() public view {
        assertEq(relay.owner(), OWNER);
        assertEq(address(relay.processor()), address(processor));
        assertEq(relay.circuitId(), CIRCUIT_ID);
        assertEq(relay.unit(), U);
        assertEq(relay.perTxCap(), 50 * U);
        assertEq(relay.dailyCap(), 150 * U);
        assertEq(relay.activationFee(), 5 * U);
    }

    function testZeroAddressesRejected() public {
        vm.expectRevert(LatchRelay.ZeroAddress.selector);
        new LatchRelay(address(0), 1, address(container), address(token), SAVINGS, TREASURY, KEEPER, 1, 1, 0, 0);
        vm.expectRevert(LatchRelay.ZeroAddress.selector);
        new LatchRelay(address(processor), 1, address(container), address(token), address(0), TREASURY, KEEPER, 1, 1, 0, 0);
        vm.expectRevert(LatchRelay.ZeroAddress.selector);
        new LatchRelay(address(processor), 1, address(container), address(token), SAVINGS, TREASURY, address(0), 1, 1, 0, 0);
    }

    function testCapOrderingRejected() public {
        vm.expectRevert(LatchRelay.OverCap.selector);
        new LatchRelay(address(processor), 1, address(container), address(token), SAVINGS, TREASURY, KEEPER, 2, 1, 0, 0);
        vm.expectRevert(LatchRelay.OverCap.selector);
        new LatchRelay(address(processor), 1, address(container), address(token), SAVINGS, TREASURY, KEEPER, 0, 1, 0, 0);
    }

    function testCodelessTokenAndProcessorRejected() public {
        vm.expectRevert(LatchRelay.BadToken.selector);
        new LatchRelay(address(processor), 1, address(container), address(0xBEEF), SAVINGS, TREASURY, KEEPER, 1, 1, 0, 0);
        vm.expectRevert(LatchRelay.BadEval.selector);
        new LatchRelay(address(0xBEEF), 1, address(container), address(token), SAVINGS, TREASURY, KEEPER, 1, 1, 0, 0);
    }

    // ---------------------------------------------------------------- circuit

    /// The taped-out netlist must allow exactly one byte: 0x7e (every bit set but paused).
    function testNetlistTruthTableAllowsOnly0x7e() public view {
        for (uint256 b = 0; b < 128; b++) {
            assertEq(relay.circuitAllows(uint8(b)), b == 0x7e, "truth table");
        }
    }

    /// Regression: the pre-fix netlist is wrong on a real processor. It must not allow 0x7e.
    function testOldNetlistIsWrongOnRealProcessor() public {
        processor.tape(99, OLD_NETLIST, 7, 1);
        bytes memory out = processor.eval(99, hex"7e");
        assertEq(uint8(out[0]) & 1, 0, "old netlist would have denied the only valid byte");
    }

    function testCircuitDenialRevertsEvenWhenSolidityPasses() public {
        Container c2 = new Container();
        container = c2;
        LatchRelay r = _deploy(address(new ConstProcessor(hex"00")), address(token));
        token.mint(address(c2), 100 * U);
        vm.prank(OWNER);
        r.arm();
        vm.prank(KEEPER);
        vm.expectRevert(abi.encodeWithSelector(LatchRelay.CircuitDenied.selector, uint8(0x7e)));
        r.tick(10 * U);
    }

    function testLyingCircuitCannotBypassSolidityCaps() public {
        Container c2 = new Container();
        container = c2;
        LatchRelay r = _deploy(address(new ConstProcessor(hex"01")), address(token));
        token.mint(address(c2), 1000 * U);
        vm.prank(OWNER);
        r.arm();
        vm.prank(KEEPER);
        vm.expectRevert(LatchRelay.OverCap.selector);
        r.tick(500 * U);
    }

    function testEmptyEvalReverts() public {
        Container c2 = new Container();
        container = c2;
        LatchRelay r = _deploy(address(new ConstProcessor("")), address(token));
        token.mint(address(c2), 100 * U);
        vm.prank(OWNER);
        r.arm();
        vm.prank(KEEPER);
        vm.expectRevert(LatchRelay.BadEval.selector);
        r.tick(10 * U);
    }

    // -------------------------------------------------------------------- arm

    function testArmPaysTreasuryFromContainer() public view {
        assertTrue(relay.armed());
        assertEq(token.balanceOf(TREASURY), 5 * U);
        assertEq(token.balanceOf(address(container)), 995 * U);
    }

    function testArmTwiceReverts() public {
        vm.prank(OWNER);
        vm.expectRevert(LatchRelay.AlreadyArmed.selector);
        relay.arm();
    }

    function testArmOnlyOwner() public {
        Container c2 = new Container();
        container = c2;
        LatchRelay r = _deploy(address(processor), address(token));
        vm.prank(KEEPER);
        vm.expectRevert(LatchRelay.NotOwner.selector);
        r.arm();
    }

    function testArmUnderfundedReverts() public {
        Container c2 = new Container();
        container = c2;
        LatchRelay r = _deploy(address(processor), address(token));
        vm.prank(OWNER);
        vm.expectRevert(LatchRelay.UnderfundedActivation.selector);
        r.arm();
    }

    function testRelayWithoutNftCannotMoveFunds() public {
        container.setNftOwner(address(0xDEAD));
        vm.prank(KEEPER);
        vm.expectRevert(bytes("not NFT owner"));
        relay.tick(10 * U);
    }

    // ------------------------------------------------------------------- tick

    function testTickSweepsAmountAndFee() public {
        vm.prank(KEEPER);
        relay.tick(50 * U);
        assertEq(token.balanceOf(SAVINGS), 50 * U);
        assertEq(token.balanceOf(TREASURY), 5 * U + (50 * U) / 100);
        assertEq(relay.spentToday(), 50 * U);
        assertEq(relay.volumeSwept(), 50 * U);
    }

    function testTickOnlyKeeper() public {
        vm.prank(OWNER);
        vm.expectRevert(LatchRelay.NotKeeper.selector);
        relay.tick(10 * U);
    }

    function testTickRequiresArming() public {
        Container c2 = new Container();
        container = c2;
        LatchRelay r = _deploy(address(processor), address(token));
        token.mint(address(c2), 100 * U);
        vm.prank(KEEPER);
        vm.expectRevert(LatchRelay.NotArmed.selector);
        r.tick(10 * U);
    }

    function testPausedTickReverts() public {
        vm.prank(OWNER);
        relay.setPaused(true);
        vm.prank(KEEPER);
        vm.expectRevert(LatchRelay.Paused.selector);
        relay.tick(10 * U);
    }

    function testZeroAndOversizedAmountsRevert() public {
        vm.startPrank(KEEPER);
        vm.expectRevert(LatchRelay.OverCap.selector);
        relay.tick(0);
        vm.expectRevert(LatchRelay.OverCap.selector);
        relay.tick(500 * U); // the console's "rogue 500 USDT"
        vm.stopPrank();
    }

    function testDailyCapAndUtcRollover() public {
        vm.startPrank(KEEPER);
        relay.tick(50 * U);
        relay.tick(50 * U);
        relay.tick(50 * U);
        vm.expectRevert(LatchRelay.OverCap.selector);
        relay.tick(1 * U);
        vm.warp(block.timestamp + 1 days);
        relay.tick(50 * U);
        vm.stopPrank();
        assertEq(relay.spentToday(), 50 * U);
    }

    function testUnderfundedIncludingFeeReverts() public {
        vm.prank(OWNER);
        relay.withdraw(OWNER, 995 * U - 50 * U); // leaves exactly 50, not 50 + fee
        vm.prank(KEEPER);
        vm.expectRevert(LatchRelay.Underfunded.selector);
        relay.tick(50 * U);
    }

    function testFalseReturningTokenReverts() public {
        FalseToken ft = new FalseToken();
        Container c2 = new Container();
        container = c2;
        LatchRelay r = _deploy(address(processor), address(ft));
        ft.mint(address(c2), 100 * U);
        vm.prank(OWNER);
        vm.expectRevert(LatchRelay.TransferFailed.selector);
        r.arm();
    }

    function testReentrancyBlocked() public {
        HookToken ht = new HookToken();
        Container c2 = new Container();
        container = c2;
        LatchRelay r = _deploy(address(processor), address(ht));
        ht.mint(address(c2), 1000 * U);
        vm.prank(OWNER);
        r.arm();
        vm.prank(OWNER);
        r.setKeeper(address(ht)); // let the hook pass the keeper check, so only the lock stops it
        ht.setRelay(address(r));
        ht.setHook(true);
        vm.prank(address(ht));
        vm.expectRevert(LatchRelay.Reentered.selector);
        r.tick(10 * U);
    }

    // ---------------------------------------------------------------- preview

    function testPreviewMatchesTickForKeeper() public {
        vm.prank(KEEPER);
        (uint8 bits, bool allow) = relay.preview(50 * U);
        assertEq(bits, 0x7e);
        assertTrue(allow);
    }

    function testPreviewFromStrangerShowsKeeperBitClear() public view {
        (uint8 bits, bool allow) = relay.preview(50 * U);
        assertEq(bits & 2, 0);
        assertFalse(allow);
    }

    function testPreviewDeniesRogueAndPaused() public {
        vm.prank(KEEPER);
        (uint8 bits, bool allow) = relay.preview(500 * U);
        assertEq(bits & 16, 0);
        assertFalse(allow);
        vm.prank(OWNER);
        relay.setPaused(true);
        vm.prank(KEEPER);
        (bits, allow) = relay.preview(10 * U);
        assertEq(bits & 1, 1);
        assertFalse(allow);
    }

    // ------------------------------------------------------------ execute fee

    function _unfunded() internal returns (LatchRelay r) {
        Container c2 = new Container();
        container = c2;
        vm.prank(OWNER);
        r = new LatchRelay(address(processor), CIRCUIT_ID, address(c2), address(token), SAVINGS, TREASURY, KEEPER, 50 * U, 150 * U, 5 * U, EF);
        c2.setNftOwner(address(r));
        token.mint(address(c2), 100 * U);
    }

    function testUnfundedRelayCannotPayContainerFee() public {
        LatchRelay r = _unfunded();
        vm.prank(OWNER);
        vm.expectRevert(abi.encodeWithSelector(LatchRelay.NoExecFee.selector, 0, EF));
        r.arm();
    }

    function testCallerCanAttachExecFee() public {
        LatchRelay r = _unfunded();
        vm.deal(OWNER, 1 ether);
        vm.prank(OWNER);
        r.arm{value: EF}();
        assertTrue(r.armed());
        vm.deal(KEEPER, 1 ether);
        uint256 cost = r.execCost(10 * U);
        assertEq(cost, 2 * EF);
        vm.prank(KEEPER);
        r.tick{value: cost}(10 * U);
        assertEq(address(r).balance, 0);
        assertEq(address(container).balance, 3 * EF);
    }

    /// A relay with execFee 0 cannot drive a fee-charging container: the container's own error surfaces.
    function testZeroExecFeeHitsContainerFee() public {
        Container c2 = new Container();
        container = c2;
        vm.prank(OWNER);
        LatchRelay r = new LatchRelay(address(processor), CIRCUIT_ID, address(c2), address(token), SAVINGS, TREASURY, KEEPER, 50 * U, 150 * U, 0, 0);
        c2.setNftOwner(address(r));
        token.mint(address(c2), 10 * U);
        vm.prank(OWNER);
        vm.expectRevert(abi.encodeWithSelector(Container.ProtocolFeeTooLow.selector, 0, EF));
        r.withdraw(OWNER, 1);
    }

    function testExecFeeBoundedAndOwnerOnly() public {
        vm.prank(KEEPER);
        vm.expectRevert(LatchRelay.NotOwner.selector);
        relay.setExecFee(1);
        vm.prank(OWNER);
        vm.expectRevert(abi.encodeWithSelector(LatchRelay.NoExecFee.selector, 0.02 ether, 0.01 ether));
        relay.setExecFee(0.02 ether);
        vm.prank(OWNER);
        relay.setExecFee(2e15);
        assertEq(relay.execFee(), 2e15);
    }

    function testOwnerRecoversOkb() public {
        uint256 bal = address(relay).balance;
        vm.prank(KEEPER);
        vm.expectRevert(LatchRelay.NotOwner.selector);
        relay.withdrawOkb(payable(KEEPER), bal);
        vm.prank(OWNER);
        relay.withdrawOkb(payable(OWNER), bal);
        assertEq(OWNER.balance, bal);
        assertEq(address(relay).balance, 0);
    }

    // ------------------------------------------------------------------ admin

    function testCapsOnlyMoveDown() public {
        vm.startPrank(OWNER);
        relay.setCaps(20 * U, 40 * U);
        vm.expectRevert(LatchRelay.CapRaised.selector);
        relay.setCaps(30 * U, 40 * U);
        vm.stopPrank();
        assertEq(relay.perTxCap(), 20 * U);
    }

    function testWithdrawOnlyOwner() public {
        vm.prank(KEEPER);
        vm.expectRevert(LatchRelay.NotOwner.selector);
        relay.withdraw(KEEPER, 1 * U);
        vm.prank(OWNER);
        relay.withdraw(OWNER, 100 * U);
        assertEq(token.balanceOf(OWNER), 100 * U);
    }

    function testTransferOwnership() public {
        vm.prank(OWNER);
        relay.transferOwnership(KEEPER);
        assertEq(relay.owner(), KEEPER);
        vm.prank(OWNER);
        vm.expectRevert(LatchRelay.NotOwner.selector);
        relay.setPaused(true);
    }

    function testStatusReadsContainerBalance() public view {
        (address t, bool armed_, bool paused_, uint256 bal,,,, address k, address s, address tr) = relay.status();
        assertEq(t, address(token));
        assertTrue(armed_);
        assertFalse(paused_);
        assertEq(bal, 995 * U);
        assertEq(k, KEEPER);
        assertEq(s, SAVINGS);
        assertEq(tr, TREASURY);
    }

    // -------------------------------------------------------------- fee ladder

    function testFeeLadderScalesWithDecimals() public view {
        assertEq(relay.feeBpsFor(0), 100);
        assertEq(relay.feeBpsFor(1_000 * U), 75);
        assertEq(relay.feeBpsFor(5_000 * U), 50);
        assertEq(relay.feeBpsFor(20_000 * U), 25);
    }

    function testFuzzTickNeverExceedsCaps(uint256 a, uint256 b) public {
        a = bound(a, 1, 50 * U);
        b = bound(b, 1, 50 * U);
        vm.startPrank(KEEPER);
        relay.tick(a);
        relay.tick(b);
        vm.stopPrank();
        assertLe(relay.spentToday(), relay.dailyCap());
        assertEq(token.balanceOf(SAVINGS), a + b);
    }
}
