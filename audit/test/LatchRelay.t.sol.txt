/// @dev Plain token with configurable decimals; tests mint freely.
contract Token {
    uint8 public immutable decimals;
    mapping(address => uint256) public balanceOf;

    constructor(uint8 decimals_) {
        decimals = decimals_;
    }

    function mint(address to, uint256 amount) external {
        balanceOf[to] += amount;
    }

    function transfer(address to, uint256 amount) external virtual returns (bool) {
        require(balanceOf[msg.sender] >= amount, "balance");
        balanceOf[msg.sender] -= amount;
        balanceOf[to] += amount;
        return true;
    }
}

/// @dev Returns false instead of reverting.
contract FalseToken is Token(6) {
    function transfer(address, uint256) external pure override returns (bool) {
        return false;
    }
}

/// @dev Calls back into the relay while it is being paid out.
contract HookToken is Token(6) {
    address public relay;
    bool public hookOn;

    function setRelay(address relay_) external {
        relay = relay_;
    }

    function setHook(bool on) external {
        hookOn = on;
    }

    function transfer(address to, uint256 amount) external override returns (bool) {
        if (hookOn) {
            hookOn = false;
            LatchRelay(relay).tick(1e6);
        }
        require(balanceOf[msg.sender] >= amount, "balance");
        balanceOf[msg.sender] -= amount;
        balanceOf[to] += amount;
        return true;
    }
}

/// @dev Stand-in for a TapeOut processor with the live X Layer TAP-02 semantics
///      (same as the processor reference checked against chain 196):
///      signal 0 = constant 0, signal 1 = constant 1, inputs at 2..2+nIn-1
///      (LSB-first from the input bytes), gate g output at 2+nIn+g, NAND only.
///      Output bit k = signal (last nOut + k), LSB-first.
contract NetlistProcessor {
    mapping(uint256 => bytes) public netlist;
    mapping(uint256 => uint32) public nInOf;
    mapping(uint256 => uint32) public nOutOf;

    function tape(uint256 id, bytes calldata net, uint32 nIn, uint32 nOut) external {
        netlist[id] = net;
        nInOf[id] = nIn;
        nOutOf[id] = nOut;
    }

    function eval(uint256 id, bytes calldata inputs) external view returns (bytes memory out) {
        bytes memory net = netlist[id];
        require(net.length > 0 && net.length % 7 == 0, "no circuit");
        uint256 nIn = nInOf[id];
        uint256 nOut = nOutOf[id];
        require(inputs.length == (nIn + 7) / 8, "bad inputs");
        uint256 n = net.length / 7;
        bool[] memory s = new bool[](2 + nIn + n);
        s[1] = true;
        for (uint256 i = 0; i < nIn; i++) {
            s[2 + i] = (uint8(inputs[i >> 3]) >> (i & 7)) & 1 == 1;
        }
        for (uint256 g = 0; g < n; g++) {
            uint256 o = g * 7;
            require(uint8(net[o]) == 0x00, "opcode");
            uint256 a = (uint256(uint8(net[o + 1])) << 16) | (uint256(uint8(net[o + 2])) << 8) | uint8(net[o + 3]);
            uint256 b = (uint256(uint8(net[o + 4])) << 16) | (uint256(uint8(net[o + 5])) << 8) | uint8(net[o + 6]);
            require(a < 2 + nIn + g && b < 2 + nIn + g, "wire");
            s[2 + nIn + g] = !(s[a] && s[b]);
        }
        out = new bytes((nOut + 7) / 8);
        uint256 base = 2 + nIn + n - nOut;
        for (uint256 k = 0; k < nOut; k++) {
            if (s[base + k]) out[k >> 3] |= bytes1(uint8(1 << (k & 7)));
        }
    }
}

/// @dev Processor that always answers the same thing (a liar or a wall).
contract ConstProcessor {
    bytes public answer;

    constructor(bytes memory answer_) {
        answer = answer_;
    }

    function eval(uint256, bytes calldata) external view returns (bytes memory) {
        return answer;
    }

    function netlist(uint256) external pure returns (bytes memory) {
        return "";
    }
}

contract LatchRelayTest is Test {
    // Circuit #1 as taped out on X Layer (processor 0x3728…6813), TAP-02 numbering.
    bytes internal constant NETLIST =
        hex"00000002000002000000030000090000000a00000a0000000b0000040000000c00000c0000000d0000050000000e00000e0000000f0000060000001000001000000011000007000000120000120000001300000800000014000014";
    // The netlist before the wire-numbering fix.
    bytes internal constant OLD_NETLIST =
        hex"000000000000000000000100000700000008000008000000090000020000000a00000a0000000b0000030000000c00000c0000000d0000040000000e00000e0000000f000005000000100000100000001100000600000012000012";
    uint256 internal constant CIRCUIT_ID = 1;
    uint256 internal constant U = 1e6; // 6-decimal USDT0, as on X Layer

    address internal constant OWNER = address(0xC0FFEE);
    address internal constant KEEPER = address(0xCAFE);
    address internal constant SAVINGS = address(0x5A7E);
    address internal constant TREASURY = address(0x7BEA5);

    LatchRelay internal relay;
    Token internal token;
    NetlistProcessor internal processor;

    function setUp() public {
        vm.warp(1_800_000_000);
        token = new Token(6);
        processor = new NetlistProcessor();
        processor.tape(CIRCUIT_ID, NETLIST, 7, 1);
        relay = _deploy(address(processor), address(token), 2 * U, 6 * U, 1 * U);
        token.mint(address(relay), 5 * U);
        vm.prank(OWNER);
        relay.arm();
    }

    function _deploy(address proc, address tok, uint256 perTx, uint256 daily, uint256 act) internal returns (LatchRelay r) {
        vm.prank(OWNER);
        r = new LatchRelay(proc, CIRCUIT_ID, tok, SAVINGS, TREASURY, KEEPER, perTx, daily, act);
    }

    function _fresh(address proc) internal returns (LatchRelay r) {
        r = _deploy(proc, address(token), 2 * U, 6 * U, 1 * U);
        token.mint(address(r), 5 * U);
    }

    // ------------------------------------------------------------ construction

    function testConstructorWiring() public view {
        assertEq(relay.owner(), OWNER);
        assertEq(address(relay.processor()), address(processor));
        assertEq(relay.circuitId(), CIRCUIT_ID);
        assertEq(relay.unit(), U);
        assertEq(relay.perTxCap(), 2 * U);
        assertEq(relay.dailyCap(), 6 * U);
        assertEq(relay.activationFee(), 1 * U);
        assertEq(relay.container(), address(relay));
    }

    function testZeroAddressesRejected() public {
        vm.expectRevert(LatchRelay.ZeroAddress.selector);
        new LatchRelay(address(0), 1, address(token), SAVINGS, TREASURY, KEEPER, 1, 1, 0);
        vm.expectRevert(LatchRelay.ZeroAddress.selector);
        new LatchRelay(address(processor), 1, address(token), address(0), TREASURY, KEEPER, 1, 1, 0);
        vm.expectRevert(LatchRelay.ZeroAddress.selector);
        new LatchRelay(address(processor), 1, address(token), SAVINGS, TREASURY, address(0), 1, 1, 0);
    }

    function testCapOrderingRejected() public {
        vm.expectRevert(LatchRelay.OverCap.selector);
        new LatchRelay(address(processor), 1, address(token), SAVINGS, TREASURY, KEEPER, 2, 1, 0);
        vm.expectRevert(LatchRelay.OverCap.selector);
        new LatchRelay(address(processor), 1, address(token), SAVINGS, TREASURY, KEEPER, 0, 1, 0);
    }

    function testCodelessTokenAndProcessorRejected() public {
        vm.expectRevert(LatchRelay.BadToken.selector);
        new LatchRelay(address(processor), 1, address(0xBEEF), SAVINGS, TREASURY, KEEPER, 1, 1, 0);
        vm.expectRevert(LatchRelay.BadEval.selector);
        new LatchRelay(address(0xBEEF), 1, address(token), SAVINGS, TREASURY, KEEPER, 1, 1, 0);
    }

    // ---------------------------------------------------------------- circuit

    function testNetlistTruthTableAllowsOnly0x7e() public view {
        for (uint256 b = 0; b < 128; b++) {
            assertEq(relay.circuitAllows(uint8(b)), b == 0x7e, "truth table");
        }
    }

    function testOldNetlistIsWrongOnRealProcessor() public {
        processor.tape(99, OLD_NETLIST, 7, 1);
        bytes memory out = processor.eval(99, hex"7e");
        assertEq(uint8(out[0]) & 1, 0);
    }

    function testCircuitDenialRevertsEvenWhenSolidityPasses() public {
        LatchRelay r = _fresh(address(new ConstProcessor(hex"00")));
        vm.prank(OWNER);
        r.arm();
        vm.prank(KEEPER);
        vm.expectRevert(abi.encodeWithSelector(LatchRelay.CircuitDenied.selector, uint8(0x7e)));
        r.tick(1 * U);
    }

    function testLyingCircuitCannotBypassSolidityCaps() public {
        LatchRelay r = _fresh(address(new ConstProcessor(hex"01")));
        vm.prank(OWNER);
        r.arm();
        vm.prank(KEEPER);
        vm.expectRevert(LatchRelay.OverCap.selector);
        r.tick(3 * U);
    }

    function testEmptyEvalReverts() public {
        LatchRelay r = _fresh(address(new ConstProcessor("")));
        vm.prank(OWNER);
        r.arm();
        vm.prank(KEEPER);
        vm.expectRevert(LatchRelay.BadEval.selector);
        r.tick(1 * U);
    }

    // -------------------------------------------------------------------- arm

    function testArmPaysTreasuryFromVault() public view {
        assertTrue(relay.armed());
        assertEq(token.balanceOf(TREASURY), 1 * U);
        assertEq(token.balanceOf(address(relay)), 4 * U);
    }

    function testArmTwiceReverts() public {
        vm.prank(OWNER);
        vm.expectRevert(LatchRelay.AlreadyArmed.selector);
        relay.arm();
    }

    function testArmOnlyOwner() public {
        LatchRelay r = _fresh(address(processor));
        vm.prank(KEEPER);
        vm.expectRevert(LatchRelay.NotOwner.selector);
        r.arm();
    }

    function testArmUnderfundedReverts() public {
        LatchRelay r = _deploy(address(processor), address(token), 2 * U, 6 * U, 1 * U);
        vm.prank(OWNER);
        vm.expectRevert(LatchRelay.UnderfundedActivation.selector);
        r.arm();
        assertFalse(r.armed());
    }

    // ------------------------------------------------------------------- tick

    function testTickSweepsAmountAndFee() public {
        vm.prank(KEEPER);
        relay.tick(2 * U);
        assertEq(token.balanceOf(SAVINGS), 2 * U);
        assertEq(token.balanceOf(TREASURY), 1 * U + (2 * U) / 100);
        assertEq(relay.spentToday(), 2 * U);
        assertEq(relay.volumeSwept(), 2 * U);
    }

    function testTickOnlyKeeper() public {
        vm.prank(OWNER);
        vm.expectRevert(LatchRelay.NotKeeper.selector);
        relay.tick(1 * U);
    }

    function testTickRequiresArming() public {
        LatchRelay r = _fresh(address(processor));
        vm.prank(KEEPER);
        vm.expectRevert(LatchRelay.NotArmed.selector);
        r.tick(1 * U);
    }

    function testPausedTickReverts() public {
        vm.prank(OWNER);
        relay.setPaused(true);
        vm.prank(KEEPER);
        vm.expectRevert(LatchRelay.Paused.selector);
        relay.tick(1 * U);
    }

    function testZeroAndRogueAmountsRevert() public {
        vm.startPrank(KEEPER);
        vm.expectRevert(LatchRelay.OverCap.selector);
        relay.tick(0);
        vm.expectRevert(LatchRelay.OverCap.selector);
        relay.tick(500 * U);
        vm.stopPrank();
    }

    function testDailyCapAndUtcRollover() public {
        token.mint(address(relay), 20 * U);
        vm.startPrank(KEEPER);
        relay.tick(2 * U);
        relay.tick(2 * U);
        relay.tick(2 * U);
        vm.expectRevert(LatchRelay.OverCap.selector);
        relay.tick(1);
        vm.warp(block.timestamp + 1 days);
        relay.tick(2 * U);
        vm.stopPrank();
        assertEq(relay.spentToday(), 2 * U);
    }

    function testUnderfundedIncludingFeeReverts() public {
        vm.prank(OWNER);
        relay.withdraw(OWNER, 2 * U); // leaves exactly 2, not 2 + fee
        vm.prank(KEEPER);
        vm.expectRevert(LatchRelay.Underfunded.selector);
        relay.tick(2 * U);
    }

    function testFalseReturningTokenReverts() public {
        FalseToken ft = new FalseToken();
        LatchRelay r = _deploy(address(processor), address(ft), 2 * U, 6 * U, 1 * U);
        ft.mint(address(r), 5 * U);
        vm.prank(OWNER);
        vm.expectRevert(LatchRelay.TransferFailed.selector);
        r.arm();
    }

    function testReentrancyBlocked() public {
        HookToken ht = new HookToken();
        LatchRelay r = _deploy(address(processor), address(ht), 2 * U, 6 * U, 1 * U);
        ht.mint(address(r), 10 * U);
        vm.prank(OWNER);
        r.arm();
        vm.prank(OWNER);
        r.setKeeper(address(ht));
        ht.setRelay(address(r));
        ht.setHook(true);
        vm.prank(address(ht));
        vm.expectRevert(LatchRelay.Reentered.selector);
        r.tick(1 * U);
    }

    // ---------------------------------------------------------------- preview

    function testPreviewMatchesTickForKeeper() public {
        vm.prank(KEEPER);
        (uint8 bits, bool allow) = relay.preview(2 * U);
        assertEq(bits, 0x7e);
        assertTrue(allow);
    }

    function testPreviewFromStrangerShowsKeeperBitClear() public view {
        (uint8 bits, bool allow) = relay.preview(2 * U);
        assertEq(bits & 2, 0);
        assertFalse(allow);
    }

    function testPreviewDeniesRogueAndPaused() public {
        vm.prank(KEEPER);
        (uint8 bits, bool allow) = relay.preview(500 * U);
        assertEq(bits & 16, 0);
        assertFalse(allow);
        vm.prank(OWNER);
        relay.setPaused(true);
        vm.prank(KEEPER);
        (bits, allow) = relay.preview(1 * U);
        assertEq(bits & 1, 1);
        assertFalse(allow);
    }

    // ------------------------------------------------------------------ admin

    function testCapsOnlyMoveDown() public {
        vm.startPrank(OWNER);
        relay.setCaps(1 * U, 3 * U);
        vm.expectRevert(LatchRelay.CapRaised.selector);
        relay.setCaps(2 * U, 3 * U);
        vm.stopPrank();
        assertEq(relay.perTxCap(), 1 * U);
    }

    function testWithdrawOnlyOwnerWorksPausedAndUnarmed() public {
        vm.prank(KEEPER);
        vm.expectRevert(LatchRelay.NotOwner.selector);
        relay.withdraw(KEEPER, 1 * U);
        vm.prank(OWNER);
        relay.setPaused(true);
        vm.prank(OWNER);
        relay.withdraw(OWNER, 4 * U);
        assertEq(token.balanceOf(OWNER), 4 * U);
        LatchRelay r = _fresh(address(processor)); // never armed
        vm.prank(OWNER);
        r.withdraw(OWNER, 5 * U);
        assertEq(token.balanceOf(OWNER), 9 * U);
    }

    function testWithdrawEvenIfProcessorBreaks() public {
        LatchRelay r = _fresh(address(new ConstProcessor("")));
        vm.prank(OWNER);
        r.withdraw(OWNER, 5 * U);
        assertEq(token.balanceOf(address(r)), 0);
    }

    function testTwoStepOwnership() public {
        vm.prank(OWNER);
        relay.transferOwnership(KEEPER);
        assertEq(relay.owner(), OWNER, "nothing changes until accepted");
        assertEq(relay.pendingOwner(), KEEPER);
        vm.prank(SAVINGS);
        vm.expectRevert(LatchRelay.NotOwner.selector);
        relay.acceptOwnership();
        vm.prank(KEEPER);
        relay.acceptOwnership();
        assertEq(relay.owner(), KEEPER);
        assertEq(relay.pendingOwner(), address(0));
        vm.prank(OWNER);
        vm.expectRevert(LatchRelay.NotOwner.selector);
        relay.setPaused(true);
    }

    function testOwnershipTypoIsRecoverable() public {
        vm.startPrank(OWNER);
        relay.transferOwnership(address(0xdead));
        relay.transferOwnership(KEEPER); // fix the typo: still owner
        relay.withdraw(OWNER, type(uint256).max);
        vm.stopPrank();
        assertEq(token.balanceOf(address(relay)), 0);
    }

    function testWithdrawMaxBeatsKeeperFrontRun() public {
        vm.prank(KEEPER);
        relay.tick(1); // lands first
        vm.prank(OWNER);
        relay.withdraw(OWNER, type(uint256).max);
        assertEq(token.balanceOf(address(relay)), 0);
    }

    function testBadRecipientsRejected() public {
        vm.expectRevert(LatchRelay.BadRecipient.selector);
        new LatchRelay(address(processor), 1, address(token), address(token), TREASURY, KEEPER, 1, 1, 0);
        vm.expectRevert(LatchRelay.BadRecipient.selector);
        new LatchRelay(address(processor), 1, address(token), SAVINGS, address(processor), KEEPER, 1, 1, 0);
    }

    function testStatusReadsVaultBalance() public view {
        (address t, bool armed_, bool paused_, uint256 bal,,,, address k, address s, address tr) = relay.status();
        assertEq(t, address(token));
        assertTrue(armed_);
        assertFalse(paused_);
        assertEq(bal, 4 * U);
        assertEq(k, KEEPER);
        assertEq(s, SAVINGS);
        assertEq(tr, TREASURY);
    }

    function testFeeLadderScalesWithDecimals() public view {
        assertEq(relay.feeBpsFor(0), 100);
        assertEq(relay.feeBpsFor(1_000 * U), 75);
        assertEq(relay.feeBpsFor(5_000 * U), 50);
        assertEq(relay.feeBpsFor(20_000 * U), 25);
    }

    function testFuzzTickNeverExceedsCaps(uint256 a, uint256 b) public {
        token.mint(address(relay), 10 * U);
        a = bound(a, 1, 2 * U);
        b = bound(b, 1, 2 * U);
        vm.startPrank(KEEPER);
        relay.tick(a);
        relay.tick(b);
        vm.stopPrank();
        assertLe(relay.spentToday(), relay.dailyCap());
        assertEq(token.balanceOf(SAVINGS), a + b);
    }
}
